# Layout focus (ADR-305). Zoom into the synthetic tree (make_synth --depth 3; which
# sibling wins follows readdir order, hence the globs and depth checks) until one cell
# covers the viewport: the focus engages, its subtree is re-squarified into the
# viewport with breadcrumb frames for its ancestors. A staged op re-projects the tree
# and recreates every interior: the focus is keyed by identity and must survive it.
# Zoom back out: the focus pops — the parent fills the viewport in its place
# (`check focuspopped` against the `mark`) — then clears at the root. Finally the same
# zoom-in with `set focus 0` must leave the focus off, for comparison shots.
resize 1200 800
set focus 1   # opt-in (ADR-305 amendment: off by default)
load $HFSGRAPH_FIXTURE 3
wait
check focus none
shot driver_focus_fit.png view
check nonblank driver_focus_fit.png
zoom 2.5 600 400
sleep 300   # the morph (220ms) settles; the idle timer restores full detail
settle
check focus */d0_*
check focusdepth 1
shot driver_focus_in.png view
check differs driver_focus_fit.png driver_focus_in.png
probe 600 400
zoom 2.5 600 400
sleep 300
settle
check focusdepth 2+
shot driver_focus_deep.png view
check differs driver_focus_in.png driver_focus_deep.png
# Re-projection: stage a trash (any file in the fixture), which rebuilds the interiors.
stage trash @f0.json
settle
check ops 1
check focusdepth 2+
shot driver_focus_staged.png view
check nonblank driver_focus_staged.png
mark
# Zoom out: the focus pops to its parent, which fills the viewport in its place; then clears.
zoom 0.4 600 400
sleep 300
settle
check focuspopped   # shallower than at the mark: the pop really happened
shot driver_focus_out.png view
check differs driver_focus_deep.png driver_focus_out.png
zoom 0.4 600 400
sleep 300
settle
check focus none
# The comparison: the zoom that engaged above must not engage with focus off.
set focus 0
zoom 2.5 600 400
sleep 300
settle
check focus none
shot driver_focus_off.png view
check differs driver_focus_in.png driver_focus_off.png
echo focus ok
quit
