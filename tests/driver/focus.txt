# Layout focus (#40, spike). Zoom into the synthetic tree (make_synth --depth 3; which
# sibling wins follows readdir order, hence the globs) until one cell covers the
# viewport: the focus engages, its subtree is re-squarified into the viewport with
# breadcrumb frames for its ancestors. Zoom back out: the focus pops to the parent,
# then clears. `set focus 0` paints the same view without it, for comparison.
resize 1200 800
load $HFSGRAPH_FIXTURE 3
wait
check focus none
shot driver_focus_fit.png view
check nonblank driver_focus_fit.png
zoom 2.5 600 400
sleep 300   # the morph (220ms) settles; the idle timer restores full detail
settle
check focus */d0_*
shot driver_focus_in.png view
check differs driver_focus_fit.png driver_focus_in.png
probe 600 400
zoom 2.5 600 400
sleep 300
settle
check focus */d0_*/d1_*
shot driver_focus_deep.png view
check differs driver_focus_in.png driver_focus_deep.png
# The same view with focus off: the canonical slivers, for comparison.
set focus 0
sleep 300
settle
check focus none
shot driver_focus_off.png view
check differs driver_focus_deep.png driver_focus_off.png
set focus 1
sleep 300
settle
check focus */d0_*/d1_*
# Zoom out: the focus pops to its parent (placed so the old focus stays put), then clears.
zoom 0.4 600 400
sleep 300
settle
check focus */d0_*
shot driver_focus_out.png view
check differs driver_focus_deep.png driver_focus_out.png
zoom 0.4 600 400
sleep 300
settle
check focus none
echo focus ok
quit
