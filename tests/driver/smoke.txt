# Driver smoke script (see src/ui/driver.h for the grammar). Run by ctest with
# QT_QPA_PLATFORM=offscreen and HFSGRAPH_FIXTURE pointing at the repo's src/ tree.
resize 1200 800
load $HFSGRAPH_FIXTURE 3
wait
check bases 1
shot driver_smoke_fit.png view
check nonblank driver_smoke_fit.png
wheel 4
settle
shot driver_smoke_gesture.png view   # mid-gesture (lazy weights frozen)
sleep 200   # let the gesture idle out
settle
shot driver_smoke_zoomed.png view
check nonblank driver_smoke_zoomed.png
check differs driver_smoke_fit.png driver_smoke_zoomed.png
check same driver_smoke_gesture.png driver_smoke_zoomed.png   # no page flip: the gesture frame is the idle frame
echo smoke ok
quit
